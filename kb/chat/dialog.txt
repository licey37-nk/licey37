# ФОРМАТ (читает ai-engine.js):
#   @rule  <имя правила>            — начало правила
#   @match <фраза1|фраза2|фраза3>   — триггеры (регистр не важен)
#   @action <тип>                   — особое поведение команды:
#            continue | stop | short | more | table | file
#   @say <текст>                    — вариант ответа (|| разделяет
#                                     несколько вариантов в строке)
#   {name} — имя ученика, {last_topic} — текущая тема
#
# ПРАВИЛА ЖИВОГО ОБЩЕНИЯ:
#   1. Отвечаем тепло, по-дружески, с эмодзи, но по делу.
#   2. Обращаемся по имени, поддерживаем разговор.
#   3. Если не знаем ответа — честно говорим и предлагаем тему.
#   4. Любые команды формата ученик понимает с первого раза.
#   5. Математику и задачки считаем встроенным калькулятором.
# ============================================================

# ---------- КОМАНДЫ УПРАВЛЕНИЯ ДИАЛОГОМ ----------

@rule cmd_continue
@match продолжай|дальше|ещё|еще|давай дальше|что там ещё|и что дальше
@action continue
@say Продолжаю, вот ещё интересное 👇||Ладно, копаю глубже 📚||Держи продолжение ⚡

@rule cmd_stop
@match стоп|остановись|хватит|понятно|всё понятно|все понятно|ясно|достаточно|заткнись
@action stop
@say Хорошо, остановились 🙂 Какую тему разобрать следующей?||Понял, ставлю на паузу ⏸️ Что спросим ещё?||Окей, закончили! Готов объяснить что-то новое 💪

@rule cmd_short
@match покороче|кратко|короче|поменьше|слишком длинно|сжато|в двух словах|тезисно
@action short
@say Сжимаю до главного 🗜️

@rule cmd_more
@match подробнее|распиши|побольше|расскажи подробнее|раскрой тему|не понял|не совсем понял|объясни лучше|развернуто
@action more
@say Разворачиваю подробнее 📖

@rule cmd_table
@match сделай таблицу|таблицей|в виде таблицы|оформи таблицей
@action table
@say Оформляю таблицей 📊

@rule cmd_compare
@match сравни|что лучше|отличие между|разница между|чем отличается
@action table
@say Сравниваю в таблице ⚖️

@rule cmd_file
@match сделай файл|дай файлом|файлом|скачать ответ|сохранить в файл|дай документ
@action file
@say Собираю ответ в файл 📥
